  <!-- ═══════════════════════════════════════════════════════════════════
       애드센스 — 고치는 곳은 아래 window.ADS 한 군데뿐이다.
       구글 스크립트 태그도 이 client 값으로 자동 생성되므로,
       ca-pub 을 두 번 적을 필요가 없다. (예전엔 여기서 어긋나 광고가 안 떴다)

       1) client : 애드센스 → 계정 → 게시자 ID (ca-pub- 로 시작하는 16자리)
       2) slots  : 애드센스 → 광고 → 광고 단위 기준 → 단위 만들고 '슬롯 ID' 복사
                   (data-ad-slot 값. 채우지 않은 자리는 그냥 비어 있게 된다)

       · client 가 ca-pub- + 숫자 형식이 아니면 구글 스크립트를 아예 부르지 않는다.
         (잘못된 ID로 호출하면 애드센스 계정에 오류로 잡힌다)
       · 자동광고만 쓸 거면 client 만 넣고 slots 는 비워 두면 된다.
       · 슬롯을 손으로 붙여넣고 싶으면, 원하는 .ad-box 안에 구글이 준
         <ins class="adsbygoogle"> 코드를 그대로 넣어라. 스크립트가 비켜준다.
       ═══════════════════════════════════════════════════════════════════ -->
  <script>
    window.ADS = {
      client: 'ca-pub-9021429421997169',
      slots: {
        top:       '7524587591',  /* 헤더 아래 리더보드 · 고정 970x90        */
        topMobile: '',            /* 좁은 화면용 (320x100 또는 반응형 단위)   */
        inArticle: '',            /* 본문 중간 · 자동 삽입   인아티클 fluid   */
        bottom:    '',            /* 본문 끝, 태그 위        반응형          */
        sidebar:   '',            /* 사이드바 상단           300x250         */
        sticky:    '',            /* 사이드바 스티키         300x600         */
        infeed:    '',            /* 목록 사이 인피드        fluid           */
        footer:    ''             /* 푸터 위                 반응형          */
      },
      autoAds: false,    /* 자동광고도 함께 쓸 때 true */
      enabled: true      /* 전체를 잠시 끌 때 false    */
    };

    /* 구글 스크립트는 유효한 ID가 있을 때만 부른다 — 단일 출처 */
    (function () {
      var a = window.ADS;
      if (!a.enabled || !/^ca-pub-\d{10,}$/.test(a.client)) return;
      var s = document.createElement('script');
      s.async = true;
      s.crossOrigin = 'anonymous';
      s.src = 'https://pagead2.googlesyndication.com/pagead/js/adsbygoogle.js?client=' + a.client;
      document.head.appendChild(s);
      if (a.autoAds) {
        var m = document.createElement('meta');
        m.name = 'google-adsense-account'; m.content = a.client;
        document.head.appendChild(m);
      }
    }());
  </script>

    <script>
    (function () {
      'use strict';
      var d = document, root = d.documentElement, body = d.body;
      var $ = function (s, c) { return (c || d).querySelector(s); };
      var $$ = function (s, c) { return Array.prototype.slice.call((c || d).querySelectorAll(s)); };
      var bodyId = body.id || '';
      var isView = /tt-body-(page|entry|notice)/.test(bodyId) || !!$('#article-view');
      var isHome = /tt-body-index|tt-body-home/.test(bodyId);

      /* ── 1. 테마 전환 ─────────────────────────────── */
      (function theme() {
        var btn = $('.btn-theme');
        if (!btn) return;
        btn.addEventListener('click', function () {
          var cur = root.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
          root.setAttribute('data-theme', cur);
          try { localStorage.setItem('skin-theme', cur); } catch (e) { }
          adaptInlineColors();          /* 본문 인라인 색을 테마에 맞춘다 */
        });
      }());

      /* ── 1.5 본문 인라인 색상 흡수 ──────────────────
         티스토리 에디터로 쓴 글에는 인라인 color·background 가 박힌다
         (실측: 글 하나에 231개). 전부 라이트 기준이라 다크에서 두 방향으로 깨진다.
           · color:#111 같은 어두운 글자 → 어두운 배경 위에서 1.02:1
           · background:#fff 같은 밝은 판 → 그 위 우리 다크 글자색이 1.08:1
         발행된 글은 수정할 수 없으므로(HANDOVER §6) 스킨이 흡수한다.
         원본 style 을 보관해 두고 라이트로 돌아가면 그대로 되돌린다.       */
      function adaptInlineColors() {
        var av = $('#article-view');
        if (!av) return;
        var dark = root.getAttribute('data-theme') === 'dark';

        /* 되돌리기 — 보관해 둔 원본 style 을 복원 */
        $$('[data-skin-orig]', av).forEach(function (el) {
          var o = el.getAttribute('data-skin-orig');
          if (o) el.setAttribute('style', o); else el.removeAttribute('style');
          el.removeAttribute('data-skin-orig');
        });
        $$('[data-skin-color]', av).forEach(function (el) {
          el.setAttribute('color', el.getAttribute('data-skin-color'));
          el.removeAttribute('data-skin-color');
        });
        if (!dark) return;

        function lumOf(c) {
          var m = String(c).match(/[\d.]+/g);
          if (!m || m.length < 3) return null;
          if (m.length > 3 && Number(m[3]) < .5) return null;   /* 거의 투명 */
          var f = function (v) { v /= 255; return v <= .03928 ? v / 12.92 : Math.pow((v + .055) / 1.055, 2.4); };
          return .2126 * f(+m[0]) + .7152 * f(+m[1]) + .0722 * f(+m[2]);
        }
        function keep(el) {
          if (!el.hasAttribute('data-skin-orig'))
            el.setAttribute('data-skin-orig', el.getAttribute('style') || '');
        }

        /* 1단계 — 밝은 인라인 배경을 어두운 면으로 바꾼다.
           먼저 배경을 정리해야 2단계에서 글자색을 판단할 수 있다. */
        $$('[style]', av).forEach(function (el) {
          var cs = getComputedStyle(el);
          if (cs.backgroundImage && cs.backgroundImage !== 'none') return;
          var L = lumOf(cs.backgroundColor);
          if (L === null || L < .55) return;                    /* 어두운 판은 그대로 */
          keep(el);
          el.style.backgroundColor = 'var(--c-bg-inset)';
          var bl = lumOf(cs.borderTopColor);
          if (cs.borderTopWidth !== '0px' && bl !== null && bl > .5) el.style.borderColor = 'var(--c-line)';
          var blf = lumOf(cs.borderLeftColor);
          if (cs.borderLeftWidth !== '0px' && blf !== null && blf > .8) el.style.borderLeftColor = 'var(--c-brand)';
        });

        /* 2단계 — 어두운 인라인 글자색을 걷어낸다.
           지우면 스킨 CSS 가 이어받아 제목은 --c-ink, 본문은 --c-ink-2 가 된다. */
        $$('[style]', av).forEach(function (el) {
          if (!el.style.color) return;
          var L = lumOf(getComputedStyle(el).color);
          if (L === null || L > .4) return;                     /* 이미 밝은 글자는 둔다 */
          var bgL = lumOf(getComputedStyle(el).backgroundColor);
          if (bgL !== null && bgL > .55) return;                /* 밝은 판 위면 짝이 맞다 */
          keep(el);
          el.style.color = '';
        });

        /* 3단계 — 구식 <font color> */
        $$('font[color]', av).forEach(function (el) {
          var L = lumOf(getComputedStyle(el).color);
          if (L === null || L > .4) return;
          el.setAttribute('data-skin-color', el.getAttribute('color'));
          el.removeAttribute('color');
        });
      }
      adaptInlineColors();

      /* ── 7.5 썸네일 정규화 ─────────────────────────
         [##_..._thumbnail_raw_url_##] 는 스킴 없는 경로나 파일명만 줄 때가 있다.
         그대로 쓰면 목록 이미지칸이 통째로 빈다. 여기서 세 가지를 한다.
           1) 스킴 보정 (//… , fname 만 온 경우 → 티스토리 리사이저 경유)
           2) img 를 정면으로 세우고 background 는 폴백으로 남긴다
           3) 깨진 이미지는 감춰서 '엑스박스' 대신 빈 카드로 만든다        */
      (function thumbs() {
        /* 티스토리 썸네일 URL 을 정리한다.
           _raw_url_ 은 서명이 붙은 원본이다 (1200px, ?credential=…&expires=…&signature=…).
           카드는 300~470px 이므로 리사이저에 태워 750px 로 줄인다.

           리사이저 규격 — 실측으로 확정한 것(다섯 형식을 브라우저에서 직접 로드해 봤다):
             원본 그대로                      ✓ 1200×1200
             fname=encodeURIComponent(원본)   ✓  750×750   ← 이것만 맞다
             fname=원본 (인코딩 안 함)         ✗ 실패  (서명의 & 가 쿼리를 끊는다)
             쿼리 떼고 인코딩                  ✗ 실패  (서명이 빠져 거부된다)
           티스토리 자신도 인코딩해서 쓴다.

           그래도 리사이저가 실패할 수 있으므로 원본으로 한 번 되돌아간다.
           두 번 다 실패하면 빈 판을 그린다.                                    */
        var RESIZE = 'https://img1.daumcdn.net/thumb/R750x0/?scode=mtistory2&fname=';
        var CDN = /^https?:\/\/(blog\.kakaocdn\.net|[a-z0-9-]+\.daumcdn\.net)\//i;

        function absolutize(u) {
          if (!u) return '';
          u = String(u).trim().replace(/^url\((['"]?)([\s\S]*?)\1\)$/, '$2').trim();
          if (!u || u === 'none') return '';
          if (/^data:/.test(u)) return u;
          if (/^\/\//.test(u)) return 'https:' + u;
          if (/^\//.test(u)) return location.origin + u;
          if (!/^https?:\/\//.test(u)) return '';   /* 정체불명 — 손대지 않는다 */
          return u;
        }

        function resized(u) {
          if (!u || /^data:/.test(u)) return u;
          if (/\/thumb\//.test(u)) return u;        /* 이미 리사이저를 거쳤다 */
          if (!CDN.test(u)) return u;               /* 외부 이미지는 그대로 */
          return RESIZE + encodeURIComponent(u);
        }

        function blank(box, img) {
          box.style.backgroundImage = '';
          box.classList.add('is-empty');
          if (img) img.setAttribute('data-broken', '');
        }

        /* 되는 이미지를 먼저 지우지 않는다.
           티스토리가 인라인으로 넣어준 배경은 이미 화면에 떠 있다.
           줄인 URL 이 확실히 로드된 뒤에만 바꿔 끼운다.
           (그 순서를 어겼더니 썸네일이 떴다가 회색판에 덮였다 — 실측) */
        function swapWhenReady(box, img, url, onFail) {
          var probe = new Image();
          probe.onload = function () {
            box.style.backgroundImage = "url('" + url + "')";
            if (img) {
              img.removeAttribute('data-broken');
              if (img.getAttribute('src') !== url) img.setAttribute('src', url);
            }
          };
          probe.onerror = function () { onFail && onFail(); };
          probe.src = url;
        }

        $$('.thumbnail').forEach(function (box) {
          var img = $('.img-thumbnail', box);
          var bg = '';
          try { bg = getComputedStyle(box).backgroundImage; } catch (e) { }
          var origin = absolutize(img && img.getAttribute('src')) || absolutize(bg);

          if (!origin) {
            if (box.hasAttribute('has-thumbnail')) blank(box, img);
            return;
          }

          if (img) {
            img.setAttribute('decoding', 'async');
            if (!img.getAttribute('loading')) img.setAttribute('loading', 'lazy');
          }

          var small = resized(origin);
          if (small === origin) {
            /* 줄일 것이 없다 — 원본이 실패하면 그때 빈 판 */
            swapWhenReady(box, img, origin, function () { blank(box, img); });
            return;
          }
          /* 줄인 것 → 실패하면 원본 → 그것도 실패하면 빈 판 */
          swapWhenReady(box, img, small, function () {
            swapWhenReady(box, img, origin, function () { blank(box, img); });
          });
        });
      }());

      /* ── 8. 본문 표 가로스크롤 · 이미지 지연로딩 ──── */
      (function content() {
        var av = $('#article-view'); if (!av) return;
        $$('table', av).forEach(function (tb) {
          if (tb.parentNode.classList.contains('table-scroll')) return;
          var w = d.createElement('div'); w.className = 'table-scroll';
          tb.parentNode.insertBefore(w, tb); w.appendChild(tb);
        });
        $$('img', av).forEach(function (im) {
          if (!im.getAttribute('loading')) im.setAttribute('loading', 'lazy');
          im.setAttribute('decoding', 'async');
        });
        $$('iframe', av).forEach(function (f) {
          if (!f.getAttribute('loading')) f.setAttribute('loading', 'lazy');
          if (f.parentNode.classList.contains('embed-ratio')) return;
          var w = d.createElement('div'); w.className = 'embed-ratio';
          f.parentNode.insertBefore(w, f); w.appendChild(f);
        });
        $$('a[href^="http"]', av).forEach(function (a) {
          if (a.hostname !== location.hostname) { a.target = '_blank'; a.rel = 'noopener nofollow'; }
        });
      }());

      /* ── 9. 목록 보기 방식 (카드형 / 목록형) ──────── */

      /* ── 11. 사이드바 드로어 ───────────────────────
         common.js 가 .area-aside 에 .area-aside-on 을, 딤에 .bg-dimmed 를 붙인다.
         여기서는 그 상태를 읽어 aria / 스크롤잠금 / ESC 만 얹는다. */
      (function drawer() {
        var aside = $('.area-aside'), btn = $('.button-menu'), dim = $('.drawer-dim');
        if (!aside || !btn) return;
        function isOpen() { return aside.classList.contains('area-aside-on'); }
        function sync() {
          var open = isOpen();
          btn.setAttribute('aria-expanded', String(open));
          btn.setAttribute('aria-label', open ? '메뉴 닫기' : '메뉴 열기');
          body.classList.toggle('drawer-open', open);
          if (dim) dim.hidden = !open;
        }
        if (window.MutationObserver) {
          new MutationObserver(sync).observe(aside, { attributes: true, attributeFilter: ['class'] });
        } else {
          btn.addEventListener('click', function () { setTimeout(sync, 60); });
        }
        if (dim) dim.addEventListener('click', function () { if (isOpen()) btn.click(); });
        d.addEventListener('keydown', function (e) {
          if (e.key === 'Escape' && isOpen()) btn.click();
        });
        sync();
      }());

      /* ── 12. 애드센스 슬롯 ─────────────────────────
         · client 가 유효해야만 <ins> 를 만든다
         · 손으로 붙여넣은 ins.adsbygoogle 가 이미 있으면 손대지 않는다
         · 슬롯 ID 가 없는 자리는 광고 없이 접는다 (빈 상자를 남기지 않는다) */
      (function ads() {
        var cfg = window.ADS || {};
        var slots = cfg.slots || {};
        var live = cfg.enabled !== false && /^ca-pub-\d{10,}$/.test(cfg.client || '');
        /* fixed:[w,h] 는 고정 크기 단위. 고정 단위는 줄어들지 않으므로
           들어갈 폭이 없는 화면에서는 아예 띄우지 않는다(minVw).
           반응형 단위를 만들면 fixed 를 지우고 fmt:'auto' 만 남기면 된다. */
        var FMT = {
          top:       { fixed: [970, 90], minVw: 1010, alt: 'topMobile' },
          topMobile: { fmt: 'auto', min: 100, maxVw: 1009 },
          bottom:    { fmt: 'auto', min: 280 },
          footer:    { fmt: 'auto', min: 100 },
          sidebar:   { fmt: 'auto', min: 250 },
          sticky:    { fmt: 'auto', min: 600 },
          inArticle: { fmt: 'fluid', layout: 'in-article', min: 280 },
          infeed:    { fmt: 'fluid', min: 250 }
        };
        var VW = Math.max(document.documentElement.clientWidth || 0, window.innerWidth || 0);

        function collapse(wrap, why) {
          wrap.classList.add('is-empty');
          wrap.setAttribute('aria-hidden', 'true');
          if (why && !ads.warned) {
            ads.warned = true;
            console.info('[스킨] 애드센스 — ' + why
              + ' skin.html 의 window.ADS 에서 값을 채우면 표시됩니다.');
          }
        }

        function mount(wrap, name) {
          var box = $('.ad-box', wrap);
          if (!box) return;
          var spec = FMT[name] || { fmt: 'auto', min: 200 };
          var minH = spec.fixed ? spec.fixed[1] : (spec.min || 200);

          /* 이미 사람이 붙여넣은 코드가 있으면 그대로 둔다 */
          if (box.querySelector('ins.adsbygoogle')) {
            var ph0 = $('.ad-ph', box); if (ph0) ph0.parentNode.removeChild(ph0);
            box.style.minHeight = minH + 'px';
            try { (window.adsbygoogle = window.adsbygoogle || []).push({}); } catch (e) { }
            return;
          }

          /* 고정 크기 단위는 폭이 되는 화면에서만. 아니면 대체 슬롯에 넘긴다 */
          if (spec.minVw && VW < spec.minVw) { collapse(wrap, ''); return; }
          if (spec.maxVw && VW > spec.maxVw) { collapse(wrap, ''); return; }

          var slot = String(slots[name] || '').replace(/\D/g, '');
          if (!live) { collapse(wrap, 'client(게시자 ID)가 설정되지 않았습니다.'); return; }
          if (!slot) { collapse(wrap, 'slots.' + name + ' 이 비어 있습니다.'); return; }

          var ph = $('.ad-ph', box); if (ph) ph.parentNode.removeChild(ph);
          box.style.minHeight = minH + 'px';

          var ins = d.createElement('ins');
          ins.className = 'adsbygoogle';
          ins.setAttribute('data-ad-client', cfg.client);
          ins.setAttribute('data-ad-slot', slot);
          if (spec.fixed) {
            /* 고정 단위: data-ad-format 을 주면 안 된다 (크기가 무시된다) */
            ins.style.display = 'inline-block';
            ins.style.width = spec.fixed[0] + 'px';
            ins.style.height = spec.fixed[1] + 'px';
          } else {
            ins.style.display = 'block';
            ins.setAttribute('data-ad-format', spec.fmt);
            if (spec.layout) ins.setAttribute('data-ad-layout', spec.layout);
            if (spec.fmt === 'auto') ins.setAttribute('data-full-width-responsive', 'true');
          }
          box.appendChild(ins);
          try { (window.adsbygoogle = window.adsbygoogle || []).push({}); } catch (e) { }
        }

        function makeSlot(cls, name) {
          var w = d.createElement('div');
          w.className = 'ad ' + cls;
          w.setAttribute('data-ad', name);
          w.innerHTML = '<span class="ad-label">Advertisement</span>'
                      + '<div class="ad-box"><span class="ad-ph">' + name + '</span></div>';
          return w;
        }

        /* 본문 중간 — 두 번째 h2 앞, 없으면 중간 문단 앞 */
        (function inArticle() {
          var av = $('#article-view'); if (!av) return;
          var hs = $$('h2', av);
          var anchor = hs.length >= 2 ? hs[1] : null;
          if (!anchor) {
            var ps = $$('p', av).filter(function (p) { return p.textContent.trim().length > 40; });
            if (ps.length >= 6) anchor = ps[Math.floor(ps.length / 2)];
          }
          if (!anchor || !anchor.parentNode) return;
          anchor.parentNode.insertBefore(makeSlot('ad-inarticle', 'inArticle'), anchor);
        }());

        /* 목록 인피드 — 세 번째 카드 뒤 */
        (function inFeed() {
          var wrap = $('.list-wrap'); if (!wrap) return;
          var cards = $$('article', wrap).filter(function (a) { return a.parentNode === wrap; });
          if (cards.length < 5) return;
          wrap.insertBefore(makeSlot('ad-infeed', 'infeed'), cards[3]);
        }());

        /* 전부(정적 + 방금 만든 것) 장착 */
        $$('.ad[data-ad]').forEach(function (a) { mount(a, a.getAttribute('data-ad')); });
      }());
